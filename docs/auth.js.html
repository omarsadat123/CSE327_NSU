/**
 * @fileoverview User authentication model.
 * Handles database queries related to user retrieval and creation.
 * Utilizes MySQL connection pool and bcrypt for password verification.
 */

const db = require('../configs/db');
const bcrypt = require('bcryptjs');

module.exports = {
  /**
   * Retrieve a user by their email address.
   *
   * @async
   * @param {string} email - The email address of the user.
   * @returns {Promise<Object|null>} Resolves with user data if found, otherwise null.
   */
  async findUserByEmail(email) {
    const [rows] = await db.query(
      'SELECT uid, name, email FROM users WHERE email = ?',
      [email]
    );
    return rows.length ? rows[0] : null;
  },

  /**
   * Retrieve a user by email and verify their password.
   *
   * @async
   * @param {string} email - The email address of the user.
   * @param {string} plainPassword - The plain-text password to verify.
   * @returns {Promise<Object|null>} Resolves with user data if email exists and password matches, otherwise null.
   */
  async findUserByEmailAndPassword(email, plainPassword) {
    const [userRows] = await db.query(
      'SELECT uid, name, email FROM users WHERE email = ?',
      [email]
    );
    if (userRows.length === 0) {
      return null;
    }

    const user = userRows[0];

    const [credRows] = await db.query(
      'SELECT password FROM credentials WHERE uid = ?',
      [user.uid]
    );
    if (credRows.length === 0) {
      return null;
    }

    const hashedPassword = credRows[0].password;
    const isMatch = await bcrypt.compare(plainPassword, hashedPassword);
    if (!isMatch) {
      return null;
    }

    return user;
  },

  /**
   * Create a new user and store their hashed password in the database.
   *
   * @async
   * @param {string} name - The name of the new user.
   * @param {string} email - The email address of the new user.
   * @param {string} hashedPassword - The hashed password of the new user.
   * @returns {Promise<Object>} Resolves with the created user's basic information.
   * @throws {Error} Throws an error if the database transaction fails.
   */
  async createUserWithCredentials(name, email, hashedPassword) {
    const conn = await db.getConnection();
    try {
      await conn.beginTransaction();

      const [userResult] = await conn.query(
        'INSERT INTO users (name, email) VALUES (?, ?)',
        [name, email]
      );
      const uid = userResult.insertId;

      await conn.query(
        'INSERT INTO credentials (uid, password) VALUES (?, ?)',
        [uid, hashedPassword]
      );

      await conn.commit();

      return { uid, name, email };
    } catch (err) {
      await conn.rollback();
      throw err;
    } finally {
      conn.release();
    }
  }
};
