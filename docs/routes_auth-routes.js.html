/**
 * @fileoverview Authentication Routes
 * 
 * This file defines the Express.js routes for user authentication-related
 * actions such as login, signup, OTP verification, and logout.
 * 
 * All route handlers are delegated to the authentication controller.
 * 
 * @module routes/auth
 */

const express = require('express');
const router = express.Router();
const authController = require('../controllers/auth-controller');

/**
 * GET /login
 * 
 * Renders the login page where users can enter their credentials.
 * 
 * @name GET /login
 * @function
 * @memberof module:routes/auth
 * @inner
 * @param {express.Request} req - Express request object.
 * @param {express.Response} res - Express response object.
 */
router.get('/login', authController.showLoginPage);

/**
 * POST /login
 * 
 * Processes login form submission. Validates user credentials and starts
 * a session if authentication succeeds.
 * 
 * @name POST /login
 * @function
 * @memberof module:routes/auth
 * @inner
 * @param {express.Request} req - Express request object containing email and password.
 * @param {express.Response} res - Express response object.
 */
router.post('/login', authController.login);

/**
 * GET /signup
 * 
 * Displays the signup page where new users can register an account.
 * 
 * @name GET /signup
 * @function
 * @memberof module:routes/auth
 * @inner
 * @param {express.Request} req - Express request object.
 * @param {express.Response} res - Express response object.
 */
router.get('/signup', authController.showSignupPage);

/**
 * POST /signup
 * 
 * Handles signup form submission. Validates input, sends an OTP for verification,
 * and temporarily stores signup data in session.
 * 
 * @name POST /signup
 * @function
 * @memberof module:routes/auth
 * @inner
 * @param {express.Request} req - Express request object containing user registration data.
 * @param {express.Response} res - Express response object.
 */
router.post('/signup', authController.handleSignup);

/**
 * GET /verify-otp
 * 
 * Renders the OTP verification page where users can enter the received OTP
 * to complete the signup process.
 * 
 * @name GET /verify-otp
 * @function
 * @memberof module:routes/auth
 * @inner
 * @param {express.Request} req - Express request object.
 * @param {express.Response} res - Express response object.
 */
router.get('/verify-otp', authController.showOtpPage);

/**
 * POST /verify-otp
 * 
 * Processes the OTP submission. If the OTP is correct, finalizes account creation.
 * 
 * @name POST /verify-otp
 * @function
 * @memberof module:routes/auth
 * @inner
 * @param {express.Request} req - Express request object containing the OTP.
 * @param {express.Response} res - Express response object.
 */
router.post('/verify-otp', authController.handleOtpVerification);

/**
 * GET /logout
 * 
 * Logs out the current user by destroying the session and clearing cookies.
 * 
 * @name GET /logout
 * @function
 * @memberof module:routes/auth
 * @inner
 * @param {express.Request} req - Express request object.
 * @param {express.Response} res - Express response object.
 */
router.get('/logout', authController.logout);

module.exports = router;
